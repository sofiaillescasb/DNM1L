make_bound_plot <- function(mod, dat, pcx = 1, pcy = 2, alg = "Model") {
  set.seed(2024)
  
  # 1. Extract feature data and preprocessing object from model
  feature_data <- dat %>% select(-c(1, 2)) # Assumes cols 1 & 2 are non-features (e.g., id, class)
  preproc <- mod$preProcess
  
  # 2. Transform original data to PCA space
  pca_coords <- predict(preproc, feature_data)
  
  plot_df <- data.frame(
    X = pca_coords[, pcx],
    Y = pca_coords[, pcy],
    class = dat$class
  )
  
  # 3. Create grid in 2D PCA space
  x_seq <- seq(min(plot_df$X) - 1, max(plot_df$X) + 1, length.out = 100)
  y_seq <- seq(min(plot_df$Y) - 1, max(plot_df$Y) + 1, length.out = 100)
  grid <- expand.grid(X = x_seq, Y = y_seq)
  
  # Build full PCA grid (fill unselected components with 0 / mean)
  grid_full <- as.data.frame(matrix(0, nrow = nrow(grid), ncol = ncol(pca_coords)))
  colnames(grid_full) <- colnames(pca_coords)
  grid_full[, pcx] <- grid$X
  grid_full[, pcy] <- grid$Y
  
  # 4. Inverse PCA transformation back to original feature space
  rotation <- preproc$rotation
  center   <- if (!is.null(preproc$mean)) preproc$mean else rep(0, nrow(rotation))
  scale    <- if (!is.null(preproc$std)) preproc$std else rep(1, nrow(rotation))
  
  rotation_sub  <- rotation[, colnames(grid_full), drop = FALSE]
  grid_original <- as.matrix(grid_full) %*% t(rotation_sub)
  grid_original <- sweep(grid_original, 2, scale, `*`)
  grid_original <- sweep(grid_original, 2, center, `+`)
  
  grid_original <- as.data.frame(grid_original)
  colnames(grid_original) <- colnames(feature_data)
  
  # 5. Predict on original feature space
  grid$class <- predict(mod, newdata = grid_original)
  
  # 6. Plot decision regions and actual data points
  ggplot() +
    geom_tile(data = grid, aes(x = X, y = Y, fill = class), alpha = 0.3) +
    geom_point(data = plot_df, aes(x = X, y = Y, color = class), size = 2.5, stroke = 0.8) +
    scale_color_manual(values = c("#a39b9b",  "#5c95ff", "#e76f51")) +    
    scale_fill_manual(name = "Prediction", values = c("#a39b9b", "#add9f4", "#f4a261")) +
    labs(
      title = paste(alg, "- Decision Boundary (PCA Space)"),
      x = paste("Principal Component", pcx),
      y = paste("Principal Component", pcy)
    ) +
    theme_classic() +
    theme(
      plot.title = element_text(face = "bold", size = 12),
      panel.grid.minor = element_blank()
    )
}

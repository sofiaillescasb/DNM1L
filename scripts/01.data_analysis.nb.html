Visualization

long_db <- db %>%
  pivot_longer(cols = -c(ID, Group), names_to = "Predictor", values_to = "Value")

long_db %>%
  ggplot(aes(x = ID, y = Value, fill = Group)) +
  geom_boxplot() +
  facet_wrap(~ Predictor, scales = "free_y") +
  scale_fill_manual(values = c("#6996e3", "#dd5129")) +
  theme_bw()


long_db %>%
  ggplot(aes(x = Group, y = Value, fill = Group)) +
  geom_boxplot() +
  facet_wrap(~ Predictor, scales = "free_y") +
  scale_fill_manual(values = c("#6996e3", "#dd5129")) +
  theme_classic()


long_db %>%
  ggplot(aes(x = Value, fill = Group)) +
  geom_density(alpha = 0.6) +
  facet_wrap(~ Predictor, scales = "free") +
  scale_fill_manual(values = c("#6996e3", "#dd5129")) +
  theme_classic()


long_db %>%
  ggplot(aes(x = Value, fill = Group)) +
  geom_histogram(alpha = 0.6) +
  facet_wrap(~ Predictor, scales = "free") +
  scale_fill_manual(values = c("#6996e3", "#dd5129")) +
  theme_classic()
